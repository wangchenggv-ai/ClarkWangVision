function Proxy-On {
    $env:HTTPS_PROXY = "http://127.0.0.1:58319"
    $env:HTTP_PROXY = "http://127.0.0.1:58319"
}
function Proxy-Off {
    Remove-Item env:HTTPS_PROXY -ErrorAction SilentlyContinue
    Remove-Item env:HTTP_PROXY -ErrorAction SilentlyContinue
}
Proxy-Off
function Use-Claude {
    $settings = Get-Content "$env:USERPROFILE\.claude\settings.json" | ConvertFrom-Json
    $settings.env = @{}
    if ($settings.PSObject.Properties["model"]) { $settings.PSObject.Properties.Remove("model") }
    $settings | ConvertTo-Json -Depth 10 | Set-Content "$env:USERPROFILE\.claude\settings.json"
    $env:ANTHROPIC_BASE_URL = ""
    $env:ANTHROPIC_API_KEY = ""
    $env:ANTHROPIC_AUTH_TOKEN = ""
    Proxy-On
    Write-Host "Ready: Anthropic Pro (proxy on)" -ForegroundColor Green
}
